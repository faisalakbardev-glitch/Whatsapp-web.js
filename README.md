# Whatsapp-web.js
Membuat Bot pesan otomatis whatsaap

# Clone Repo
git clone https://github.com/faisalakbardev-glitch/Whatsapp-web.js.git

# Masuk ke foldernya
cd Whatsaap-web.js

# Jika sudah, Inisialisasi projek Node.js di terminal vscode/terminal biasa
npm init -y

# Install library whatsapp-web.js dan qrcode-terminal (buat scan QR nanti)
npm install whatsapp-web.js qrcode-terminal
